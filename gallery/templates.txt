# ODE App Builder templates gallery.
# Each "app" block becomes a ready-made project; tools/embed.mjs compiles it with the same form
# model the question editor uses. Format:
#   app <id>            then  key: value  lines (title, sector, blurb, icon, name, desc, primary, secondary, design, welcome, footer)
#   form <id> "Title" icon=.. desc=".." parent=<form id> summary=a,b
#   page "Title" if=<condition>
#   note "Text" if=<condition>
#   <type> <name> "Label" req ro min=.. max=.. hint=".." if=<condition> choices=a|b|c  (or value:Label|value:Label)
# Types: text longtext phone email integer decimal one many rating date time datetime adate timer photo gps signature audio video scan file
# Conditions: q=v  q!=v  q~v (includes)  q>n q>=n q<n q<=n  q? (answered); join with & (all) or , (any)

app health
title: Community health
sector: Health
icon: stethoscope
blurb: Household register, member list, home visits with danger signs and referrals, pregnancy follow-up.
name: Community Health
desc: Household visits and referrals
primary: #0F766E
secondary: #B45309
design: dashboard
welcome: Register each household once. Open it to add members, record visits and follow up pregnancies.
footer: In an emergency, refer to the nearest health facility straight away.

form household "Household" icon=house desc="Register once per household" summary=head_name,village
page "Where"
text village "Village or zone" req
text head_name "Head of household" req
phone phone "Phone number"
gps location "Location of the house"
page "The household"
integer members "How many people live here?" req min=1 max=40
one water "Main source of drinking water" req choices=Piped|Borehole|Protected well|Unprotected well|River or lake|Other
one latrine "Does the household have a latrine?" req choices=yes:Yes|no:No
integer nets "How many sleeping nets are in use?" min=0 max=20

form member "Household member" icon=user parent=household desc="Each person in the household" summary=full_name,sex
text full_name "Full name" req
one sex "Sex" req choices=female:Female|male:Male
adate dob "Date of birth" req hint="Use the year only if the exact date is not known."
one relation "Relationship to head" choices=Head|Spouse|Child|Parent|Other relative|Not related
one pregnant "Is she pregnant?" if=sex=female choices=yes:Yes|no:No|unknown:Does not know

form visit "Home visit" icon=stethoscope parent=household desc="Each time you visit" summary=visit_date,reason
date visit_date "Date of visit" req
one reason "Main reason for the visit" req choices=Routine|Sick person|Pregnancy follow-up|Newborn check|Referral follow-up
page "Danger signs"
many danger "Danger signs seen" hint="Tick all that apply." choices=None|Fever|Fast breathing|Diarrhoea with blood|Convulsions|Unable to drink|Bleeding in pregnancy
decimal temperature "Temperature (°C)" min=34 max=43 if=danger~fever
page "Action"
one referred "Referred to a health facility?" req choices=yes:Yes|no:No
text facility "Which facility?" req if=referred=yes
photo referral_slip "Photo of the referral slip" if=referred=yes
date next_visit "Next visit date"
longtext notes "Notes"
